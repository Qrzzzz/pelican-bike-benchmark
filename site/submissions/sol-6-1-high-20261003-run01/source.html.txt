  <script>
    (() => {
      'use strict';
      const byId = id => document.getElementById(id);
      const button = byId('toggle');
      const label = byId('button-label');
      const illustration = byId('illustration');
      const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
      const narrowScreen = window.matchMedia('(max-width: 640px)');
      const parts = {
        rear: byId('rear-spokes'), front: byId('front-spokes'), crank: byId('crank'),
        backLeg: byId('back-leg'), frontLeg: byId('front-leg'), highlight: byId('leg-highlight'),
        backFoot: byId('back-foot'), frontFoot: byId('front-foot'),
        backPedal: byId('back-pedal'), frontPedal: byId('front-pedal'), rider: byId('rider'),
        scarf: byId('scarf-tail'), road: byId('road-marks'), clouds: byId('clouds'),
        water: byId('water'), foreground: byId('foreground')
      };
      let playing = false;
      let elapsed = 0;
      let lastTime = null;
      let frameId = null;
      const number = value => value.toFixed(2);

      function legPath(hipX, hipY, footX, footY) {
        const dx = footX - hipX;
        const dy = footY - hipY;
        const distance = Math.max(1, Math.hypot(dx, dy));
        const upper = 112;
        const lower = 112;
        const along = (upper * upper - lower * lower + distance * distance) / (2 * distance);
        const bend = Math.sqrt(Math.max(0, upper * upper - along * along));
        const kneeX = hipX + along * dx / distance + bend * dy / distance;
        const kneeY = hipY + along * dy / distance - bend * dx / distance;
        return `M${hipX} ${hipY}L${number(kneeX)} ${number(kneeY)}L${number(footX)} ${number(footY)}`;
      }

      function render() {
        const angle = elapsed * Math.PI * 2 / 1.7;
        const degrees = angle * 180 / Math.PI;
        const pedalX = Math.cos(angle) * 29;
        const pedalY = Math.sin(angle) * 29;
        const frontX = 620 + pedalX;
        const frontY = 542 + pedalY;
        const backX = 620 - pedalX;
        const backY = 542 - pedalY;
        const frontPath = legPath(596, 374, frontX, frontY - 5);
        parts.frontLeg.setAttribute('d', frontPath);
        parts.highlight.setAttribute('d', frontPath);
        parts.backLeg.setAttribute('d', legPath(572, 369, backX, backY - 5));
        parts.frontFoot.setAttribute('transform', `translate(${number(frontX)} ${number(frontY - 3)})`);
        parts.backFoot.setAttribute('transform', `translate(${number(backX)} ${number(backY - 3)})`);
        parts.frontPedal.setAttribute('transform', `translate(${number(frontX)} ${number(frontY)})`);
        parts.backPedal.setAttribute('transform', `translate(${number(backX)} ${number(backY)})`);
        parts.crank.setAttribute('transform', `translate(620 542) rotate(${number(degrees % 360)})`);
        const wheelDegrees = (degrees * 1.7) % 360;
        parts.rear.setAttribute('transform', `rotate(${number(wheelDegrees)})`);
        parts.front.setAttribute('transform', `rotate(${number(wheelDegrees)})`);
        parts.rider.setAttribute('transform', `translate(0 ${number(Math.sin(angle * 2) * 1.5)})`);
        parts.scarf.setAttribute('transform', `rotate(${number(Math.sin(angle * 1.3) * 3)} 673 281)`);
        parts.road.setAttribute('transform', `translate(${number(-(elapsed * 105) % 220)} 0)`);
        parts.clouds.setAttribute('transform', `translate(${number(Math.sin(elapsed * .08) * 22)} 0)`);
        parts.water.setAttribute('transform', `translate(${number(-(elapsed * 7) % 1200)} 0)`);
        parts.foreground.setAttribute('transform', `translate(${number(-(elapsed * 105) % 1200)} 0)`);
      }

      function tick(time) {
        frameId = null;
        if (!playing || document.hidden) return;
        if (lastTime !== null) elapsed += Math.min((time - lastTime) / 1000, .05);
        lastTime = time;
        render();
        frameId = requestAnimationFrame(tick);
      }

      function stopFrame() {
        if (frameId !== null) cancelAnimationFrame(frameId);
        frameId = null;
        lastTime = null;
      }

      function setPlaying(value) {
        playing = value;
        document.body.dataset.playing = String(value);
        button.setAttribute('aria-pressed', String(value));
        button.setAttribute('aria-label', value ? '暂停动画' : '播放动画');
        label.textContent = value ? '暂停动画' : '播放动画';
        stopFrame();
        if (playing && !document.hidden) frameId = requestAnimationFrame(tick);
      }

      function resizeScene() {
        illustration.setAttribute('viewBox', narrowScreen.matches ? '315 40 650 650' : '0 30 1200 650');
      }

      button.addEventListener('click', () => setPlaying(!playing));
      document.addEventListener('keydown', event => {
        if (event.code !== 'Space' && event.key !== ' ') return;
        const target = event.target;
        if (target instanceof HTMLElement &&
            (target.isContentEditable || /^(INPUT|TEXTAREA|SELECT)$/.test(target.tagName))) return;
        event.preventDefault();
        if (!event.repeat) setPlaying(!playing);
      });
      document.addEventListener('visibilitychange', () => {
        stopFrame();
        if (playing && !document.hidden) frameId = requestAnimationFrame(tick);
      });
      reducedMotion.addEventListener('change', event => {
        if (event.matches) setPlaying(false);
      });
      narrowScreen.addEventListener('change', resizeScene);
      resizeScene();
      render();
      setPlaying(!reducedMotion.matches);
    })();
  </script>
