  function pedalAt(angle) {

  function solveLeg(hip, ankle) {
    var dx = ankle.x - hip.x;
    var dy = ankle.y - hip.y;
    var d = Math.sqrt(dx * dx + dy * dy) || 0.001;
    var maxD = THIGH + SHIN - 0.5;
    if (d > maxD) {
      var sc = maxD / d;
      dx *= sc;
      dy *= sc;
      d = maxD;
    }
    var a = Math.atan2(dy, dx);
    var cosA = (THIGH * THIGH + d * d - SHIN * SHIN) / (2 * THIGH * d);
    cosA = cosA > 1 ? 1 : (cosA < -1 ? -1 : cosA);
    var ka = a - Math.acos(cosA);
    return {
      knee: { x: hip.x + THIGH * Math.cos(ka), y: hip.y + THIGH * Math.sin(ka) },
      ankle: { x: hip.x + dx, y: hip.y + dy }
    };
  }

      'M' + hipF.x + ' ' + hipF.y.toFixed(1) +

  /* ---------- responsive viewBox ---------- */
  function layout() {
    var cw = stage.clientWidth || 0;
    var ch = stage.clientHeight || 0;
    if (cw < 2 || ch < 2) { return; }
    var aspect = cw / ch;
    var w, h;
    if (aspect >= 1.5) {
      h = 800;
      w = 800 * aspect;
    } else {
      w = 1200;
      h = 1200 / aspect;
    }
    if (h > 1180) { h = 1180; w = h * aspect; }
    if (w < 620) { w = 620; h = w / aspect; }
    if (w > 3000) { w = 3000; h = w / aspect; }
    svg.setAttribute('viewBox',
      (600 - w / 2).toFixed(1) + ' ' + (400 - h / 2).toFixed(1) + ' ' + w.toFixed(1) + ' ' + h.toFixed(1));
  }

  /* ---------- animation loop ---------- */
  var RAF = window.requestAnimationFrame
    ? function (fn) { return window.requestAnimationFrame(fn); }
    : function (fn) { return window.setTimeout(function () { fn(Date.now()); }, 16); };
  var CAF = window.cancelAnimationFrame ? function (id) { window.cancelAnimationFrame(id); } : window.clearTimeout;

  var SPEED = 3.2;
  var playing = false;
  var phase = 0;
  var lastT = 0;
  var rafId = 0;

  function frame(t) {
    if (!playing) { return; }
    var dt = lastT ? (t - lastT) / 1000 : 0;
    lastT = t;
    if (dt > 0.08) { dt = 0.08; }
    phase += dt * SPEED;
    if (phase > 1e6) { phase = mod(phase, Math.PI * 2); }
    render(phase);
    rafId = RAF(frame);
  }

  function setPlaying(on) {
    if (on === playing) { return; }
    playing = on;
    btn.classList.toggle('is-playing', on);
    btn.setAttribute('aria-pressed', on ? 'true' : 'false');
    btn.setAttribute('aria-label', on ? '暂停动画' : '播放动画');
    btnText.textContent = on ? '暂停' : '播放';
    if (on) {
      lastT = 0;
      rafId = RAF(frame);
    } else {
      CAF(rafId);
    }
  }

  layout();
  render(0);

  btn.addEventListener('click', function () { setPlaying(!playing); });

  document.addEventListener('keydown', function (e) {
    var isSpace = e.code === 'Space' || e.key === ' ' || e.key === 'Spacebar';
    if (!isSpace || e.repeat) { return; }
    if (e.target === btn) { return; }
    e.preventDefault();
    setPlaying(!playing);
  });

  window.addEventListener('resize', function () { RAF(layout); });
  window.addEventListener('orientationchange', function () { window.setTimeout(layout, 150); });
  window.addEventListener('load', layout);
  if (window.ResizeObserver) {
    new window.ResizeObserver(function () { layout(); }).observe(stage);
  }

  var mq = window.matchMedia ? window.matchMedia('(prefers-reduced-motion: reduce)') : null;
  function onMotionPref(ev) {
    if (ev.matches) { setPlaying(false); }
  }
  if (mq) {
    if (mq.matches) { setPlaying(false); } else { setPlaying(true); }
    if (mq.addEventListener) { mq.addEventListener('change', onMotionPref); }
    else if (mq.addListener) { mq.addListener(onMotionPref); }
  } else {
    setPlaying(true);
  }
}());
