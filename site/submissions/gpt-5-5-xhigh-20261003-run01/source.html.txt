  <script>
    (() => {
      "use strict";

      const reduceMotion = window.matchMedia("(prefers-reduced-motion: reduce)");
      const button = document.querySelector(".control");
      const label = button.querySelector(".control-label");
      const icon = button.querySelector(".control-icon");
      const riderBike = document.querySelector(".rider-bike");
      const pelican = document.querySelector(".pelican");
      const bodyGroup = document.querySelector(".body-group");
      const headGroup = document.querySelector(".head-group");
      const wingGroup = document.querySelector(".wing-group");
      const rearWheel = document.querySelector(".rear-wheel .spokes");
      const frontWheel = document.querySelector(".front-wheel .spokes");
      const crankSet = document.querySelector(".crank-set");
      const legFront = document.querySelector(".leg-front");
      const legBack = document.querySelector(".leg-back");
      const roadLines = document.querySelector(".road-lines");
      const speedLines = document.querySelector(".speed-lines");
      const grass = document.querySelector(".moving-grass");
      const parallax = document.querySelector(".parallax-slow");

      const state = {
        playing: !reduceMotion.matches,
        start: 0,
        elapsed: 0,
        frame: 0
      };

      function setTransform(element, value) {
        element.setAttribute("transform", value);
      }

      function setButton() {
        button.setAttribute("aria-pressed", String(state.playing));
        button.setAttribute("aria-label", state.playing ? "暂停动画" : "播放动画");
        label.textContent = state.playing ? "暂停" : "播放";
        icon.innerHTML = state.playing
          ? '<path d="M7 5h3.4v14H7zM13.6 5H17v14h-3.4z" fill="currentColor"></path>'
          : '<path d="M7.5 4.6v14.8L18.8 12 7.5 4.6z" fill="currentColor"></path>';
      }

      function render(ms) {
        const t = ms / 1000;
        const wheelAngle = -t * 430;
        const crankAngle = t * 360;
        const bob = Math.sin(t * Math.PI * 2) * 7;
        const sway = Math.sin(t * Math.PI * 2 + 0.7) * 2.4;
        const wing = Math.sin(t * Math.PI * 2 + 0.4) * 5;
        const head = Math.sin(t * Math.PI * 2 + 1.1) * 3.5;
        const ground = -(t * 180 % 260);
        const grassX = -(t * 96 % 160);
        const slow = -(t * 14 % 90);
        const dash = -(t * 260 % 280);

        setTransform(rearWheel, `rotate(${wheelAngle})`);
        setTransform(frontWheel, `rotate(${wheelAngle})`);
        setTransform(crankSet, `translate(650 650) rotate(${crankAngle})`);
        setTransform(riderBike, `translate(0 ${bob}) rotate(${sway} 620 535)`);
        setTransform(pelican, `translate(0 ${Math.sin(t * Math.PI * 2 + 0.3) * 3})`);
        setTransform(bodyGroup, `rotate(${sway * 0.7} 638 446)`);
        setTransform(headGroup, `rotate(${head} 900 305)`);
        setTransform(wingGroup, `rotate(${wing} 596 443)`);
        setTransform(legFront, `rotate(${Math.sin(t * Math.PI * 2 + Math.PI) * 9} 650 650)`);
        setTransform(legBack, `rotate(${Math.sin(t * Math.PI * 2) * 9} 650 650)`);
        setTransform(roadLines, `translate(${ground} 0)`);
        setTransform(grass, `translate(${grassX} 0)`);
        setTransform(parallax, `translate(${slow} 0)`);
        setTransform(speedLines, `translate(${dash} 0)`);
      }

      function step(now) {
        if (!state.playing) {
          state.frame = 0;
          return;
        }

        if (!state.start) {
          state.start = now - state.elapsed;
        }

        state.elapsed = now - state.start;
        render(state.elapsed);
        state.frame = window.requestAnimationFrame(step);
      }

      function play() {
        if (state.playing) {
          return;
        }
        state.playing = true;
        state.start = 0;
        setButton();
        state.frame = window.requestAnimationFrame(step);
      }

      function pause() {
        if (!state.playing) {
          return;
        }
        state.playing = false;
        window.cancelAnimationFrame(state.frame);
        setButton();
      }

      function toggle() {
        if (state.playing) {
          pause();
        } else {
          play();
        }
      }

      button.addEventListener("click", toggle);

      window.addEventListener("keydown", (event) => {
        const tag = event.target && event.target.tagName;
        const typing = tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT" || event.target?.isContentEditable;
        if (event.code === "Space" && !typing) {
          event.preventDefault();
          toggle();
        }
      });

      reduceMotion.addEventListener("change", (event) => {
        if (event.matches) {
          pause();
        } else if (!state.playing && state.elapsed === 0) {
          play();
        }
      });

      setButton();
      render(0);

      if (state.playing) {
        state.frame = window.requestAnimationFrame(step);
      }
    })();
  </script>
