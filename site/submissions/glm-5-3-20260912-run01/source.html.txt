      <svg class="ic icPause" viewBox="0 0 24 24" aria-hidden="true"><rect x="6" y="5" width="4" height="14" rx="1.4"/><rect x="14" y="5" width="4" height="14" rx="1.4"/></svg>
      <span id="btnLabel">播放</span>
    </button>
    <p class="hint">空格键或点按按钮：播放 / 暂停</p>
  </div>
</main>

<script>
(function () {
  'use strict';

  var $ = function (id) { return document.getElementById(id); };
  var scene = $('scene'), playBtn = $('playBtn'), btnLabel = $('btnLabel'), status = $('status');
  var wR = $('wR'), wF = $('wF'), cN = $('cN'), cF = $('cF');
  var pN = $('pN'), pF = $('pF'), footN = $('footN'), footF = $('footF');
  var legNI = $('legNI'), legFI = $('legFI');
  var birdA = $('birdA'), birdB = $('birdB'), wing = $('wing'), eye = $('eye');
  var chT = $('chT'), chB = $('chB'), cl = $('cl');

  var DEG = Math.PI / 180;
  var GROUND = 250;            // 地面前进速度 px/s
  var WDEG = (GROUND / 82) * 180 / Math.PI;  // 车轮角速度 deg/s
  var CRW = 324;               // 曲柄角速度 deg/s（约 0.9 转/秒）
  var TH0 = -50;               // 初始曲柄相位
  var BBX = 612, BBY = 624, CR = 34;
  var HIP_N = { x: 648, y: 462 }, HIP_F = { x: 626, y: 466 };
  var L1 = 116, L2 = 112;
  var WRAP = 2400, LO = -560;  // 背景循环区间 [-560, 1840)

  var f1 = function (v) { return Math.round(v * 10) / 10; };
  var f2 = function (v) { return Math.round(v * 100) / 100; };
  var T = function (el, tr) { el.setAttribute('transform', tr); };

  // ---- 背景视差元素登记 ----
  var movers = [];
  var nodes = scene.querySelectorAll('[data-px]');
  for (var i = 0; i < nodes.length; i++) {
    var el = nodes[i];
    movers.push({ el: el, x0: parseFloat(el.getAttribute('data-x')), y: parseFloat(el.getAttribute('data-y')), f: parseFloat(el.getAttribute('data-px')) });
  }

  // ---- 幅条与气门嘴 ----
  function buildSpokes() {
    var NS = 'http://www.w3.org/2000/svg';
    var spec = [['spR', 468, 583], ['spF', 792, 583]];
    spec.forEach(function (s) {
      var g = $(s[0]), cx = s[1], cy = s[2], k, a, l;
      for (k = 0; k < 8; k++) {
        a = k * Math.PI / 4;
        l = document.createElementNS(NS, 'line');
        l.setAttribute('x1', f1(cx + 12 * Math.cos(a)));
        l.setAttribute('y1', f1(cy + 12 * Math.sin(a)));
        l.setAttribute('x2', f1(cx + 66 * Math.cos(a)));
        l.setAttribute('y2', f1(cy + 66 * Math.sin(a)));
        l.setAttribute('stroke', '#cfd9e6');
        l.setAttribute('stroke-width', '2.5');
        g.appendChild(l);
      }
      l = document.createElementNS(NS, 'line');
      l.setAttribute('x1', f1(cx + 8)); l.setAttribute('y1', f1(cy + 64));
      l.setAttribute('x2', f1(cx + 10)); l.setAttribute('y2', f1(cy + 76));
      l.setAttribute('stroke', '#3f4552');
      l.setAttribute('stroke-width', '4');
      l.setAttribute('stroke-linecap', 'round');
      g.appendChild(l);
    });
  }

  // ---- 两骨 IK：返回靠前（+x）的膝盖 ----
  function ik(h, ft) {
    var dx = ft.x - h.x, dy = ft.y - h.y;
    var d = Math.max(8, Math.min(224, Math.hypot(dx, dy)));
    var a = Math.atan2(dy, dx);
    var c = Math.min(1, Math.max(-1, (L1 * L1 + d * d - L2 * L2) / (2 * L1 * d)));
    var b = Math.acos(c);
    var k1x = h.x + L1 * Math.cos(a + b), k1y = h.y + L1 * Math.sin(a + b);
    var k2x = h.x + L1 * Math.cos(a - b), k2y = h.y + L1 * Math.sin(a - b);
    return (k1x >= k2x) ? { x: k1x, y: k1y } : { x: k2x, y: k2y };
  }
  function drawLeg(path, h, ft) {
    var k = ik(h, ft);
    path.setAttribute('d', 'M ' + f1(h.x) + ' ' + f1(h.y) +
      ' L ' + f1(k.x) + ' ' + f1(k.y) +
      ' L ' + f1(ft.x) + ' ' + f1(ft.y));
  }

  // ---- 每帧渲染 ----
  var render = function (t) {
    var th = (TH0 + CRW * t) % 360;
    var thr = th * DEG, phr = thr + Math.PI;
    var wd = (WDEG * t) % 360;

    T(wR, 'rotate(' + f1(wd) + ' 468 583)');
    T(wF, 'rotate(' + f1(wd) + ' 792 583)');
    T(cN, 'rotate(' + f1(th) + ' 612 624)');
    T(cF, 'rotate(' + f1(th + 180) + ' 612 624)');

    var pnx = BBX + CR * Math.cos(thr), pny = BBY + CR * Math.sin(thr);
    var pfx = BBX + CR * Math.cos(phr), pfy = BBY + CR * Math.sin(phr);
    T(pN, 'translate(' + f1(pnx) + ' ' + f1(pny) + ')');
    T(pF, 'translate(' + f1(pfx) + ' ' + f1(pfy) + ')');
    T(footN, 'translate(' + f1(pnx) + ' ' + f1(pny - 12) + ')');
    T(footF, 'translate(' + f1(pfx) + ' ' + f1(pfy - 12) + ')');

    var by = 2.4 * Math.sin(2 * thr);
    var br = Math.sin(thr);
    var tr = 'translate(0 ' + f1(by) + ') rotate(' + f1(br) + ' 585 380)';
    T(birdA, tr); T(birdB, tr);
    T(wing, 'rotate(' + f1(2.4 * Math.sin(t * 2.027)) + ' 676 330)');

    var s = 1, bp = t % 3.7;
    if (bp > 3.56) { s = 1 - 0.92 * Math.sin(Math.PI * ((bp - 3.56) / 0.14)); }
    T(eye, 'translate(786 268) scale(1 ' + f2(Math.max(0.06, s)) + ') translate(-786 -268)');

    drawLeg(legNI, { x: HIP_N.x, y: HIP_N.y + by }, { x: pnx, y: pny - 26 });
    drawLeg(legFI, { x: HIP_F.x, y: HIP_F.y + by }, { x: pfx, y: pfy - 26 });

    var o16 = (118.75 * t) % 16;
    chT.setAttribute('stroke-dashoffset', f2(o16));
    chB.setAttribute('stroke-dashoffset', f2(16 - o16));
    cl.setAttribute('stroke-dashoffset', f2((GROUND * t) % 86));

    for (var j = 0; j < movers.length; j++) {
      var m = movers[j];
      var x = m.x0 - GROUND * t * m.f;
      x = ((x - LO) % WRAP + WRAP) % WRAP + LO;
      m.el.setAttribute('transform', 'translate(' + f1(x) + ' ' + m.y + ')');
    }
  };

  // ---- 播放控制 ----
  var playing = false, t = 0, lastTs = 0, raf = 0;

  function setUI() {
    document.body.setAttribute('data-play', playing ? '1' : '0');
    playBtn.setAttribute('aria-pressed', playing ? 'true' : 'false');
    btnLabel.textContent = playing ? '暂停' : '播放';
    status.textContent = playing ? '状态：前进中' : '状态：已暂停';
  }
  function loop(ts) {
    if (!playing) return;
    if (lastTs) t += Math.min(0.05, (ts - lastTs) / 1000);
    lastTs = ts;
    render(t);
    raf = requestAnimationFrame(loop);
  }
  function play() {
    if (playing) return;
    playing = true; lastTs = 0; setUI();
    raf = requestAnimationFrame(loop);
  }
  function pause() {
    if (!playing) return;
    playing = false;
    cancelAnimationFrame(raf);
    setUI();
  }
  function toggle() { playing ? pause() : play(); }

  var lastSpace = 0;
  playBtn.addEventListener('click', function () {
    if (performance.now() - lastSpace < 300) return;
    toggle();
  });
  window.addEventListener('keydown', function (e) {
    if (e.code === 'Space') {
      e.preventDefault();
      if (e.repeat) return;
      lastSpace = performance.now();
      toggle();
    }
  });

  // ---- 视口适配：动态 viewBox，保证小车完整可见 ----
  function onResize() {
    var w = window.innerWidth, h = window.innerHeight;
    if (!w || !h) return;
    var a = w / h;
    var H = 1200 / a;
    H = Math.min(1400, Math.max(430, H));
    var W = a * H;
    var x0 = 660 - W / 2;
    var minX = -560, maxX = 1840 - W;
    if (maxX < minX) maxX = minX;
    x0 = Math.min(Math.max(x0, minX), maxX);
    var y0 = 800 - H;
    scene.setAttribute('viewBox',
      f1(x0) + ' ' + f1(y0) + ' ' + f1(W) + ' ' + f1(H));
  }

  buildSpokes();
  onResize();
  window.addEventListener('resize', onResize);

  render(0);

  var mq = window.matchMedia('(prefers-reduced-motion: reduce)');
  if (mq.addEventListener) {
    mq.addEventListener('change', function (e) { if (e.matches && playing) pause(); });
  }
  if (!mq.matches) { play(); } else { setUI(); }
})();
</script>
</body>
</html>
