  <script>
    (() => {
      'use strict';

      const byId = (id) => document.getElementById(id);
      const art = byId('art');
      const button = byId('motionToggle');
      const label = byId('motionLabel');
      const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
      const rearSpokes = byId('rearSpokes');
      const frontSpokes = byId('frontSpokes');
      const crankArms = byId('crankArms');
      const frontPedal = byId('frontPedal');
      const backPedal = byId('backPedal');
      const frontLeg = byId('frontLeg');
      const backLeg = byId('backLeg');
      const frontFoot = byId('frontFoot');
      const backFoot = byId('backFoot');
      const pelican = byId('pelican');
      const wing = byId('wing');
      const roadMarkings = byId('roadMarkings');
      const clouds = [byId('cloudOne'), byId('cloudTwo'), byId('cloudThree')];

      let playing = !reducedMotion.matches;
      let userChangedPlayback = false;
      let elapsed = 0;
      let previousTime = 0;
      let frameId = 0;

      function fitScene() {
        const width = window.innerWidth;
        const height = window.innerHeight;
        const mobile = width <= 700;
        const worldWidth = mobile ? 790 : 1200;
        const worldHeight = worldWidth * height / width;
        const left = mobile ? 240 : 0;
        const top = mobile ? 575 - worldHeight * .6 : 400 - worldHeight * .5;
        art.setAttribute('viewBox', `${left} ${top} ${worldWidth} ${worldHeight}`);
      }

      function drawLeg(path, foot, hipX, hipY, pedalX, pedalY, front, bob, phase) {
        const kneeX = hipX + (pedalX - hipX) * (front ? .48 : .38) + (front ? 15 : -20) + Math.sin(phase) * 8;
        const kneeY = (front ? 463 : 473) + Math.cos(phase) * 11 + bob * .5;
        path.setAttribute('d', `M${hipX} ${hipY + bob} L${kneeX.toFixed(1)} ${kneeY.toFixed(1)} L${pedalX.toFixed(1)} ${pedalY.toFixed(1)}`);
        foot.setAttribute('d', `M${(pedalX - 10).toFixed(1)} ${(pedalY + 2).toFixed(1)} Q${(pedalX + 6).toFixed(1)} ${(pedalY + 10).toFixed(1)} ${(pedalX + 24).toFixed(1)} ${(pedalY + 3).toFixed(1)}`);
      }

      function render(time) {
        const wheelDegrees = (time * 176) % 360;
        const crank = time * 3.8 - .65;
        const crankDegrees = crank * 180 / Math.PI;
        const pedalRadius = 56;
        const frontX = 630 + Math.cos(crank) * pedalRadius;
        const frontY = 575 + Math.sin(crank) * pedalRadius;
        const backX = 630 - Math.cos(crank) * pedalRadius;
        const backY = 575 - Math.sin(crank) * pedalRadius;
        const bob = Math.sin(time * 7.6) * 3;

        rearSpokes.setAttribute('transform', `rotate(${wheelDegrees})`);
        frontSpokes.setAttribute('transform', `rotate(${wheelDegrees})`);
        crankArms.setAttribute('transform', `rotate(${crankDegrees} 630 575)`);
        frontPedal.setAttribute('transform', `translate(${frontX.toFixed(1)} ${frontY.toFixed(1)})`);
        backPedal.setAttribute('transform', `translate(${backX.toFixed(1)} ${backY.toFixed(1)})`);
        drawLeg(frontLeg, frontFoot, 586, 386, frontX, frontY, true, bob, crank);
        drawLeg(backLeg, backFoot, 548, 387, backX, backY, false, bob, crank + Math.PI);
        pelican.setAttribute('transform', `translate(0 ${bob.toFixed(1)})`);
        wing.setAttribute('transform', `rotate(${(Math.sin(time * 3.8) * 1.3).toFixed(2)} 565 316)`);
        roadMarkings.setAttribute('patternTransform', `translate(${(-time * 210) % 185} 0)`);
        clouds[0].setAttribute('transform', `translate(${(Math.sin(time * .38) * 18).toFixed(1)} 0)`);
        clouds[1].setAttribute('transform', `translate(${(Math.sin(time * .29 + 1) * 12).toFixed(1)} 0)`);
        clouds[2].setAttribute('transform', `translate(${(Math.sin(time * .32 + 2) * 15).toFixed(1)} 0)`);
      }

      function tick(now) {
        if (!playing || document.hidden) return;
        if (previousTime) elapsed += Math.min((now - previousTime) / 1000, .05);
        previousTime = now;
        render(elapsed);
        frameId = requestAnimationFrame(tick);
      }

      function syncPlayback() {
        button.setAttribute('aria-pressed', String(playing));
        label.textContent = playing ? '暂停动画' : '播放动画';
        if (frameId) cancelAnimationFrame(frameId);
        frameId = 0;
        previousTime = 0;
        if (playing && !document.hidden) frameId = requestAnimationFrame(tick);
      }

      function togglePlayback() {
        userChangedPlayback = true;
        playing = !playing;
        syncPlayback();
      }

      button.addEventListener('click', togglePlayback);
      document.addEventListener('keydown', (event) => {
        if (event.code !== 'Space' && event.key !== ' ') return;
        event.preventDefault();
        if (!event.repeat) togglePlayback();
      });
      document.addEventListener('visibilitychange', syncPlayback);
      window.addEventListener('resize', fitScene, { passive: true });
      reducedMotion.addEventListener('change', () => {
        if (!userChangedPlayback) {
          playing = !reducedMotion.matches;
          syncPlayback();
        }
      });

      fitScene();
      render(0);
      syncPlayback();
    })();
  </script>
</body>
