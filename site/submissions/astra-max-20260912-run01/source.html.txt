        <button id="toggle" class="toggle" type="button" aria-controls="coastal-scene" aria-label="播放动画" aria-keyshortcuts="Space">
          <svg class="play-icon" viewBox="0 0 16 16" aria-hidden="true"><path d="M4 2.5a.8.8 0 0 1 1.2-.7l8 5.5a.8.8 0 0 1 0 1.4l-8 5.5a.8.8 0 0 1-1.2-.7Z"/></svg>
          <svg class="pause-icon" viewBox="0 0 16 16" aria-hidden="true"><rect x="3" y="2" width="3.5" height="12" rx="1"/><rect x="9.5" y="2" width="3.5" height="12" rx="1"/></svg>
          <span id="button-label">播放动画</span>
        </button>
        <span class="key-hint"><kbd>空格</kbd>也可以播放 / 暂停</span>
      </div>
      <div class="endnote"><span class="status-dot" aria-hidden="true"></span><span id="motion-note">靠岸歇一会儿</span></div>
    </footer>
    <p id="announcement" class="sr-only" role="status" aria-live="polite"></p>
    <noscript><p>启用 JavaScript 即可播放动画。你仍然可以欣赏这幅静态海岸插画。</p></noscript>
  </main>

  <script>
    (() => {
      'use strict';
      const byId = id => document.getElementById(id);
      const button = byId('toggle');
      const label = byId('button-label');
      const note = byId('motion-note');
      const announcement = byId('announcement');
      const preference = window.matchMedia('(prefers-reduced-motion: reduce)');
      const mobile = window.matchMedia('(max-width: 600px)');
      const scene = byId('coastal-scene');
      const parts = Object.fromEntries([
        'rear-spokes', 'front-spokes', 'crank', 'near-pedal', 'far-pedal',
        'near-leg', 'near-leg-outline', 'far-leg', 'far-leg-outline',
        'pelican', 'scarf-tail', 'road-marks', 'roadside-plants', 'water-lines',
        'clouds', 'gulls', 'sailboat', 'breeze', 'bike-shadow'
      ].map(id => [id, byId(id)]));
      let playing = false;
      let manualChoice = false;
      let frame = null;
      let lastTime = null;
      let elapsed = 0;
      const TAU = Math.PI * 2;
      const transform = (id, value) => parts[id].setAttribute('transform', value);

      // Equal-length leg segments keep both webbed feet on the moving pedals.
      function legPath(hipX, hipY, footX, footY, length) {
        const dx = footX - hipX;
        const dy = footY - hipY;
        const distance = Math.hypot(dx, dy);
        const bend = Math.sqrt(Math.max(0, length * length - distance * distance / 4));
        const kneeX = (hipX + footX) / 2 + dy / distance * bend;
        const kneeY = (hipY + footY) / 2 - dx / distance * bend;
        return `M${hipX.toFixed(2)} ${hipY.toFixed(2)}L${kneeX.toFixed(2)} ${kneeY.toFixed(2)}L${footX.toFixed(2)} ${footY.toFixed(2)}`;
      }

      function draw(time) {
        const angle = time * TAU / 2.2 - .6;
        const bob = Math.sin(time * TAU / 1.1) * 1.6;
        const pedalX = Math.cos(angle) * 32;
        const pedalY = Math.sin(angle) * 32;
        transform('rear-spokes', `rotate(${time * 140})`);
        transform('front-spokes', `rotate(${time * 140})`);
        transform('crank', `rotate(${angle * 180 / Math.PI})`);
        transform('near-pedal', `translate(${547 + pedalX} ${416 + pedalY})`);
        transform('far-pedal', `translate(${547 - pedalX} ${416 - pedalY})`);
        const nearLeg = legPath(519, 289 + bob, 547 + pedalX, 410 + pedalY, 86);
        const farLeg = legPath(505, 292 + bob, 547 - pedalX, 410 - pedalY, 86);
        parts['near-leg'].setAttribute('d', nearLeg);
        parts['near-leg-outline'].setAttribute('d', nearLeg);
        parts['far-leg'].setAttribute('d', farLeg);
        parts['far-leg-outline'].setAttribute('d', farLeg);
        transform('pelican', `translate(0 ${bob})`);
        transform('scarf-tail', `rotate(${Math.sin(time * 4) * 3.3} 550 191)`);
        transform('road-marks', `translate(${-((time * 215) % 280)} 0)`);
        transform('roadside-plants', `translate(${-((time * 160) % 1120)} 0)`);
        transform('water-lines', `translate(${-((time * 20) % 400)} 0)`);
        transform('clouds', `translate(${Math.sin(time / 15) * 24} 0)`);
        transform('gulls', `translate(${Math.sin(time / 8) * 12} ${Math.sin(time / 2) * 3})`);
        transform('sailboat', `translate(${Math.sin(time / 12) * 9} ${Math.sin(time * 1.2) * 1.4})`);
        transform('breeze', `translate(${Math.sin(time * 2) * 8} 0)`);
        parts['bike-shadow'].setAttribute('rx', String(219 + bob * 1.2));
      }

      function tick(timestamp) {
        if (!playing || document.hidden) { frame = null; return; }
        if (lastTime !== null) elapsed += Math.min(timestamp - lastTime, 64) / 1000;
        lastTime = timestamp;
        draw(elapsed);
        frame = requestAnimationFrame(tick);
      }

      function syncClock() {
        if (frame !== null) cancelAnimationFrame(frame);
        frame = null;
        lastTime = null;
        if (playing && !document.hidden) frame = requestAnimationFrame(tick);
      }

      function setPlaying(value, announce = false) {
        playing = value;
        document.body.dataset.playing = String(playing);
        const action = playing ? '暂停动画' : '播放动画';
        label.textContent = action;
        button.setAttribute('aria-label', action);
        note.textContent = playing ? '海风正好，慢慢向前' : preference.matches && !manualChoice ? '已尊重减少动态偏好' : '靠岸歇一会儿';
        if (announce) announcement.textContent = playing ? '动画已播放。' : '动画已暂停。';
        syncClock();
      }

      function toggle() { manualChoice = true; setPlaying(!playing, true); }
      button.addEventListener('click', toggle);
      document.addEventListener('keydown', event => {
        if ((event.code !== 'Space' && event.key !== ' ') || event.altKey || event.ctrlKey || event.metaKey || event.isComposing) return;
        const target = event.target instanceof Element ? event.target : null;
        if (target && (target.isContentEditable || target.closest('input, textarea, select, a, [role="textbox"]'))) return;
        if (target && target.closest('button') && target.closest('button') !== button) return;
        event.preventDefault();
        if (!event.repeat) toggle();
      });
      document.addEventListener('visibilitychange', syncClock);
      preference.addEventListener('change', () => {
        if (!manualChoice) setPlaying(!preference.matches);
      });

      // On phones, frame the complete rider more closely while preserving scale.
      function fitScene() {
        scene.setAttribute('viewBox', mobile.matches ? '305 0 560 550' : '0 0 1120 550');
        scene.setAttribute('preserveAspectRatio', mobile.matches ? 'xMidYMid meet' : 'xMidYMid slice');
      }
      mobile.addEventListener('change', fitScene);
      fitScene();
      draw(0);
      setPlaying(!preference.matches);
    })();
  </script>
</body>
