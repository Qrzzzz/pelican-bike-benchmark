  <script>
    (() => {
      const scene = document.getElementById('scene');
      const toggle = document.getElementById('motionToggle');
      const label = document.getElementById('motionLabel');
      const glyph = document.getElementById('motionGlyph');
      const status = document.getElementById('motionStatus');
      const motionQuery = window.matchMedia('(prefers-reduced-motion: reduce)');
      let isPlaying = !motionQuery.matches;

      function renderMotionState() {
        scene.classList.toggle('is-paused', !isPlaying);
        scene.classList.toggle('motion-allowed', isPlaying);
        toggle.setAttribute('aria-pressed', String(isPlaying));
        toggle.setAttribute('aria-label', isPlaying ? '暂停动画' : '播放动画');
        label.textContent = isPlaying ? '暂停动画' : '播放动画';
        glyph.className = `motion-glyph ${isPlaying ? 'is-pause' : 'is-play'}`;
        status.textContent = isPlaying ? '动画已播放' : '动画已暂停';
      }

      function toggleMotion() {
        isPlaying = !isPlaying;
        renderMotionState();
      }

      toggle.addEventListener('click', toggleMotion);

      window.addEventListener('keydown', (event) => {
        const activeElement = document.activeElement;
        const isFormField = activeElement && (
          ['INPUT', 'TEXTAREA', 'SELECT'].includes(activeElement.tagName) ||
          activeElement.isContentEditable
        );

        if (event.code !== 'Space' || event.repeat || isFormField) {
          return;
        }

        event.preventDefault();
        // Prevent the browser's native button activation and own the shortcut,
        // so Space toggles exactly once whether the button is focused or not.
        toggleMotion();
      });

      motionQuery.addEventListener?.('change', (event) => {
        if (event.matches) {
          isPlaying = false;
          renderMotionState();
        }
      });

      renderMotionState();
    })();
  </script>
