  <script>
    (() => {
      const root = document.documentElement;
      const button = document.getElementById('motion-toggle');
      const label = document.getElementById('motion-label');
      const status = document.getElementById('motion-status');
      const scene = document.getElementById('scene');
      const rider = document.getElementById('rider-composition');
      const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
      let playing = !reduceMotion.matches;
      let userHasChosen = false;
      let spaceOnButton = false;

      function renderState() {
        root.classList.toggle('motion-paused', !playing);
        button.setAttribute('aria-pressed', String(playing));
        button.setAttribute('aria-label', playing ? '暂停动画' : '播放动画');
        label.textContent = playing ? '暂停动画' : '播放动画';
        status.textContent = playing ? '动画播放中' : '动画已暂停';
        if (playing) scene.unpauseAnimations();
        else scene.pauseAnimations();
      }
      function setPlaying(value, userAction = false) {
        playing = value;
        if (userAction) userHasChosen = true;
        renderState();
      }
      button.addEventListener('click', () => setPlaying(!playing, true));
      document.addEventListener('keydown', event => {
        if (event.code !== 'Space' || event.repeat) return;
        const target = event.target;
        if (target instanceof HTMLElement && (target.isContentEditable || target.closest('input, textarea, select, [role="textbox"]'))) return;
        event.preventDefault();
        if (target === button || button.contains(target)) spaceOnButton = true;
        else setPlaying(!playing, true);
      });
      document.addEventListener('keyup', event => {
        if (event.code !== 'Space') return;
        event.preventDefault();
        if (spaceOnButton) {
          spaceOnButton = false;
          setPlaying(!playing, true);
        }
      });
      reduceMotion.addEventListener('change', event => {
        if (!userHasChosen) setPlaying(!event.matches);
      });

      function fitScene() {
        if (window.matchMedia('(max-width: 680px)').matches) {
          scene.setAttribute('viewBox', '310 0 580 850');
          rider.setAttribute('transform', 'translate(600 425) scale(.9) translate(-600 -425)');
          scene.setAttribute('preserveAspectRatio', 'xMidYMid meet');
        } else {
          scene.setAttribute('viewBox', '0 0 1200 800');
          scene.setAttribute('preserveAspectRatio', 'xMidYMid slice');
          rider.removeAttribute('transform');
        }
      }
      fitScene();
      window.addEventListener('resize', fitScene, { passive: true });
      renderState();
    })();
  </script>
