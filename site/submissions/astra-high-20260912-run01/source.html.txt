<script>
(() => {
  'use strict';
  const byId = id => document.getElementById(id);
  const ids = ['rear-spokes','front-spokes','crank','near-leg','far-leg','near-foot','far-foot','near-pedal','far-pedal','rider','scarf-tail','road','sea','clouds','breeze'];
  const parts = Object.fromEntries(ids.map(id => [id, byId(id)]));
  const button = byId('toggle');
  const preference = matchMedia('(prefers-reduced-motion: reduce)');
  const narrow = matchMedia('(max-width: 600px)');
  let playing = false;
  let frame = null;
  let previous = null;
  let elapsed = 0;
  const transform = (id, value) => parts[id].setAttribute('transform', value);

  function resize() {
    byId('scene').setAttribute('viewBox', narrow.matches ? '285 45 605 565' : '0 0 1100 650');
  }
  function leg(side, phase, bob) {
    const x = 540 + Math.cos(phase) * 32;
    const y = 461 + Math.sin(phase) * 32;
    const hx = side === 'near' ? 532 : 525;
    const hy = 330 + bob;
    const dx = x - hx, dy = y - hy;
    const distance = Math.hypot(dx, dy);
    const bend = Math.sqrt(Math.max(0, 86 * 86 - distance * distance / 4));
    const kx = (hx + x) / 2 + dy / distance * bend;
    const ky = (hy + y) / 2 - dx / distance * bend;
    parts[side + '-leg'].setAttribute('d', `M${hx} ${hy}L${kx} ${ky} ${x} ${y}`);
    parts[side + '-foot'].setAttribute('d', `M${x-6} ${y-7}Q${x+1} ${y-12} ${x+7} ${y-5}L${x+22} ${y+1} ${x+10} ${y+3} ${x+3} ${y+1} ${x-8} ${y+3}Z`);
    transform(side + '-pedal', `translate(${x} ${y})`);
  }
  function draw() {
    const phase = elapsed * Math.PI * 2 / 1.65;
    const bob = Math.sin(phase * 2) * 1.6;
    const rotation = (elapsed * 210) % 360;
    transform('rear-spokes', `rotate(${rotation})`);
    transform('front-spokes', `rotate(${rotation})`);
    transform('crank', `rotate(${phase * 180 / Math.PI % 360} 540 461)`);
    transform('rider', `translate(0 ${bob})`);
    transform('scarf-tail', `rotate(${Math.sin(phase) * 3} 570 210)`);
    transform('road', `translate(${-elapsed * 110 % 210} 0)`);
    transform('sea', `translate(${-elapsed * 12 % 170} 0)`);
    transform('clouds', `translate(${Math.sin(elapsed * .18) * 16} 0)`);
    parts.breeze.setAttribute('opacity', .45 + Math.sin(elapsed * 2) * .2);
    leg('far', phase + Math.PI, bob);
    leg('near', phase, bob);
  }
  function tick(now) {
    frame = null;
    if (!playing || document.hidden) return;
    if (previous !== null) elapsed += Math.min((now - previous) / 1000, .05);
    previous = now;
    draw();
    frame = requestAnimationFrame(tick);
  }
  function schedule() {
    if (frame !== null) cancelAnimationFrame(frame);
    frame = null;
    previous = null;
    if (playing && !document.hidden) frame = requestAnimationFrame(tick);
  }
  function setPlaying(value, announce = true) {
    playing = value;
    button.setAttribute('aria-pressed', String(playing));
    button.setAttribute('aria-label', playing ? '暂停动画' : '播放动画');
    byId('control-label').textContent = playing ? '暂停骑行' : '播放骑行';
    byId('control-icon').setAttribute('d', playing ? 'M3 2h4v12H3ZM10 2h4v12h-4Z' : 'M4 2l10 6-10 6Z');
    if (announce) byId('status').textContent = playing ? '骑行动画已播放' : '骑行动画已暂停';
    schedule();
  }
  button.addEventListener('click', () => setPlaying(!playing));
  document.addEventListener('keydown', event => {
    if (event.code !== 'Space' || event.altKey || event.ctrlKey || event.metaKey) return;
    if (event.target instanceof Element && event.target.closest('input,textarea,select,[contenteditable="true"]')) return;
    event.preventDefault();
    if (!event.repeat) setPlaying(!playing);
  });
  document.addEventListener('visibilitychange', schedule);
  preference.addEventListener('change', event => { if (event.matches) setPlaying(false); });
  narrow.addEventListener('change', resize);
  resize();
  draw();
  setPlaying(!preference.matches, false);
})();
</script>
