  <script>
    (() => {
      const button = document.getElementById('motionToggle');
      const label = document.getElementById('controlLabel');
      const sceneArt = document.getElementById('sceneArt');
      const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
      const narrowViewport = window.matchMedia('(max-width: 700px)');
      const shortViewport = window.matchMedia('(max-height: 500px)');
      let isPlaying = !reducedMotion.matches;

      const renderFraming = () => {
        if (narrowViewport.matches && !shortViewport.matches) {
          sceneArt.setAttribute('viewBox', '170 0 860 800');
          sceneArt.setAttribute('preserveAspectRatio', 'xMidYMid meet');
        } else if (shortViewport.matches) {
          sceneArt.setAttribute('viewBox', '0 0 1200 800');
          sceneArt.setAttribute('preserveAspectRatio', 'xMidYMid meet');
        } else {
          sceneArt.setAttribute('viewBox', '0 0 1200 800');
          sceneArt.setAttribute('preserveAspectRatio', 'xMidYMid slice');
        }
      };

      const renderState = () => {
        document.body.classList.toggle('is-playing', isPlaying);
        button.setAttribute('aria-pressed', String(isPlaying));
        button.setAttribute('aria-label', isPlaying ? '暂停动画' : '播放动画');
        label.textContent = isPlaying ? '暂停' : '播放';
      };

      const toggleMotion = () => {
        isPlaying = !isPlaying;
        renderState();
      };

      button.addEventListener('click', toggleMotion);

      document.addEventListener('keydown', (event) => {
        if (event.code !== 'Space' || event.altKey || event.ctrlKey || event.metaKey) return;

        const activeElement = document.activeElement;
        const tagName = activeElement?.tagName;
        const isTyping = tagName === 'INPUT' || tagName === 'TEXTAREA' || tagName === 'SELECT' || activeElement?.isContentEditable;
        if (isTyping) return;

        event.preventDefault();
        toggleMotion();
      });

      reducedMotion.addEventListener?.('change', (event) => {
        if (event.matches && isPlaying) {
          isPlaying = false;
          renderState();
        }
      });

      narrowViewport.addEventListener?.('change', renderFraming);
      shortViewport.addEventListener?.('change', renderFraming);

      renderFraming();
      renderState();
    })();
  </script>
