<script>
(function() {
  'use strict';

  const svg = document.getElementById('scene');
  const btn = document.getElementById('toggle-btn');
  const btnIcon = document.getElementById('btn-icon');
  const btnText = document.getElementById('btn-text');

  // Elements
  const rearSpokes = document.getElementById('rear-spokes');
  const frontSpokes = document.getElementById('front-spokes');
  const crankArm = document.getElementById('crank-arm');
  const pedal = document.getElementById('pedal');
  const body = document.getElementById('body');
  const tail = document.getElementById('tail');
  const wing = document.getElementById('wing');
  const neck = document.getElementById('neck');
  const head = document.getElementById('head');
  const eye = document.getElementById('eye');
  const beakUpper = document.getElementById('beak-upper');
  const beakLower = document.getElementById('beak-lower');
  const pouch = document.getElementById('pouch');
  const legUpper = document.getElementById('leg-upper');
  const legLower = document.getElementById('leg-lower');
  const foot = document.getElementById('foot');

  const roadDashes = Array.from(document.querySelectorAll('#road-dashes rect'));
  const bush1 = document.getElementById('bush1');
  const bush2 = document.getElementById('bush2');
  const bush3 = document.getElementById('bush3');
  const cloud1 = document.getElementById('cloud1');
  const cloud2 = document.getElementById('cloud2');

  // Base positions
  const BBX = 510, BBY = 568;
  const CRANK_R = 30;
  const REAR_X = 360, REAR_Y = 562;
  const FRONT_X = 720, FRONT_Y = 562;
  const HIP_X = 485, HIP_BASE_Y = 430;
  const BODY_BASE_Y = 385;
  const HEAD_BASE_Y = 338;
  const PIVOT_X = 540, PIVOT_BASE_Y = 395;

  // Road dash bases
  const dashBases = roadDashes.map(r => parseFloat(r.getAttribute('x')));
  const PERIOD = 200;
  const WRAP = 1400;

  // Bush bases
  let bushBases = [
    {g: bush1, base: 95, factor: 0.32},
    {g: bush2, base: 410, factor: 0.32},
    {g: bush3, base: 990, factor: 0.32}
  ];
  const bushWrap = 1250;

  // Cloud bases
  let cloudData = [
    {g: cloud1, base: 220, factor: 0.06},
    {g: cloud2, base: 780, factor: 0.06}
  ];
  const cloudWrap = 1350;

  // State
  let playing = true;
  let angle = 12; // deg
  let lastTime = performance.now();

  function rotatePoint(px, py, cx, cy, deg) {
    const rad = deg * Math.PI / 180;
    const dx = px - cx;
    const dy = py - cy;
    // clockwise to match SVG rotate
    return [
      cx + dx * Math.cos(rad) + dy * Math.sin(rad),
      cy - dx * Math.sin(rad) + dy * Math.cos(rad)
    ];
  }

  function updateScene() {
    const rad = angle * Math.PI / 180;
    const speedFactor = 0.96;

    // Wheel spokes rotation (clockwise)
    const rot = `rotate(${angle} 0 0)`;
    rearSpokes.setAttribute('transform', rot);
    frontSpokes.setAttribute('transform', rot);

    // Pedal / crank position
    const pedalX = BBX + CRANK_R * Math.sin(rad);
    const pedalY = BBY + CRANK_R * Math.cos(rad) * 0.92;

    crankArm.setAttribute('x2', pedalX);
    crankArm.setAttribute('y2', pedalY);
    pedal.setAttribute('x', pedalX - 15);
    pedal.setAttribute('y', pedalY - 3);

    // Bob & flap
    const bob = Math.sin(rad * 1.15) * 3.6;
    const headBob = Math.sin(rad * 2.4) * 1.8;
    const flap = Math.sin(rad * 3.6) * 16;

    // Body bob
    body.setAttribute('cy', BODY_BASE_Y + bob);

    // Tail bob
    const tY = bob * 0.85;
    tail.setAttribute('d', `M428,${395 + tY} Q395,${415 + tY} 405,${455 + tY} Q428,${432 + tY} 445,${400 + tY}`);

    // Wing: shift + rotate points
    const wBob = bob * 0.55;
    const wPivotY = PIVOT_BASE_Y + wBob;
    let wPts = [
      [535, 372 + wBob],
      [592, 358 + wBob],
      [625, 388 + wBob],
      [598, 418 + wBob],
      [535, 395 + wBob]
    ];
    const rotated = wPts.map(p => rotatePoint(p[0], p[1], PIVOT_X, wPivotY, flap));
    wing.setAttribute('points', rotated.map(p => p[0].toFixed(1) + ',' + p[1].toFixed(1)).join(' '));

    // Neck
    const neckStartY = 355 + bob;
    const neckEndY = 338 + headBob;
    neck.setAttribute('d', `M555,${neckStartY.toFixed(1)} Q580,${((neckStartY + neckEndY) / 2 - 4).toFixed(1)} 600,${neckEndY.toFixed(1)}`);

    // Head & face
    head.setAttribute('cy', HEAD_BASE_Y + headBob);
    eye.setAttribute('cy', 333 + headBob);

    const hb = headBob;
    beakUpper.setAttribute('d', `M632,${(336 + hb).toFixed(1)} L728,${(340 + hb).toFixed(1)}`);
    beakLower.setAttribute('d', `M632,${(341 + hb).toFixed(1)} L722,${(347 + hb).toFixed(1)}`);

    // Pouch swing
    const pouchSwing = Math.sin(rad * 1.7) * 4.5;
    const pY = 348 + hb;
    pouch.setAttribute('d', `M645,${pY.toFixed(1)} Q${(682 + pouchSwing * 0.3).toFixed(1)},${(372 + pouchSwing).toFixed(1)} 718,${(352 + hb * 0.6).toFixed(1)}`);

    // Legs (simple IK approximation)
    const hipY = HIP_BASE_Y + bob;
    const footX = pedalX + 2;
    const footY = pedalY + 4;

    // 2-segment leg approx
    const L1 = 52, L2 = 48;
    const dx = footX - HIP_X;
    const dy = footY - hipY;
    const dist = Math.hypot(dx, dy) || 0.001;
    let kx, ky;

    if (dist >= L1 + L2) {
      const r = L1 / (L1 + L2);
      kx = HIP_X + dx * r;
      ky = hipY + dy * r;
    } else {
      const cosA = (L1 * L1 + dist * dist - L2 * L2) / (2 * L1 * dist);
      const a = Math.acos(Math.max(-1, Math.min(1, cosA)));
      const baseA = Math.atan2(dy, dx);
      const kneeA = baseA - a;
      kx = HIP_X + L1 * Math.cos(kneeA);
      ky = hipY + L1 * Math.sin(kneeA);
    }

    legUpper.setAttribute('x1', HIP_X);
    legUpper.setAttribute('y1', hipY);
    legUpper.setAttribute('x2', kx);
    legUpper.setAttribute('y2', ky);

    legLower.setAttribute('x1', kx);
    legLower.setAttribute('y1', ky);
    legLower.setAttribute('x2', footX);
    legLower.setAttribute('y2', footY);

    foot.setAttribute('cx', footX);
    foot.setAttribute('cy', footY);

    // Road dashes
    const roadOffset = angle * speedFactor;
    for (let i = 0; i < roadDashes.length; i++) {
      let x = dashBases[i] - roadOffset;
      x = ((x % WRAP) + WRAP) % WRAP;
      roadDashes[i].setAttribute('x', x);
    }

    // Bushes (parallax)
    for (let i = 0; i < bushBases.length; i++) {
      const b = bushBases[i];
      let bx = b.base - (roadOffset * b.factor);
      bx = ((bx % bushWrap) + bushWrap) % bushWrap;
      const circles = b.g.querySelectorAll('circle');
      const dx = bx - b.base;
      circles.forEach(c => {
        const baseCx = parseFloat(c.getAttribute('data-base') || c.getAttribute('cx'));
        if (!c.hasAttribute('data-base')) c.setAttribute('data-base', baseCx);
        c.setAttribute('cx', baseCx + dx);
      });
    }

    // Clouds (slow)
    for (let i = 0; i < cloudData.length; i++) {
      const c = cloudData[i];
      let cx = c.base - (roadOffset * c.factor);
      cx = ((cx % cloudWrap) + cloudWrap) % cloudWrap;
      const dx = cx - c.base;
      const ells = c.g.querySelectorAll('ellipse');
      ells.forEach(e => {
        const baseCx = parseFloat(e.getAttribute('data-base') || e.getAttribute('cx'));
        if (!e.hasAttribute('data-base')) e.setAttribute('data-base', baseCx);
        e.setAttribute('cx', baseCx + dx);
      });
    }
  }

  function togglePlay() {
    playing = !playing;
    if (playing) {
      lastTime = performance.now();
      btnIcon.textContent = '⏸';
      btnText.textContent = '暂停';
      btn.setAttribute('aria-pressed', 'true');
    } else {
      btnIcon.textContent = '▶';
      btnText.textContent = '播放';
      btn.setAttribute('aria-pressed', 'false');
    }
  }

  btn.addEventListener('click', togglePlay);

  document.addEventListener('keydown', function(e) {
    if (e.key === ' ' || e.key === 'Spacebar' || e.key === 'Space') {
      e.preventDefault();
      togglePlay();
    }
  });

  // Reduced motion
  const reduceMQ = window.matchMedia('(prefers-reduced-motion: reduce)');
  if (reduceMQ.matches) {
    playing = false;
    btnIcon.textContent = '▶';
    btnText.textContent = '播放';
    btn.setAttribute('aria-pressed', 'false');
  }

  reduceMQ.addEventListener('change', function() {
    if (reduceMQ.matches && playing) {
      playing = false;
      btnIcon.textContent = '▶';
      btnText.textContent = '播放';
      btn.setAttribute('aria-pressed', 'false');
    }
  });

  // Main loop
  function loop(now) {
    if (playing) {
      const dt = now - lastTime;
      lastTime = now;
      angle = (angle + dt * 0.185) % 360;
      updateScene();
    } else {
      lastTime = now;
    }
    requestAnimationFrame(loop);
  }

  // Boot
  updateScene();
  lastTime = performance.now();
  requestAnimationFrame(loop);

  // Initial button state
  if (playing) {
    btnIcon.textContent = '⏸';
    btnText.textContent = '暂停';
    btn.setAttribute('aria-pressed', 'true');
  } else {
    btnIcon.textContent = '▶';
    btnText.textContent = '播放';
    btn.setAttribute('aria-pressed', 'false');
  }

  // Keyboard focus hint for button
  btn.setAttribute('tabindex', '0');
})();
</script>
