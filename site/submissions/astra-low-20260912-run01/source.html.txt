  <script>
    (() => {
      "use strict";

      const button = document.getElementById("toggle");
      const label = document.getElementById("button-label");
      const preference = window.matchMedia("(prefers-reduced-motion: reduce)");
      const parts = Object.fromEntries([
        "near-leg", "near-leg-highlight", "far-leg", "near-foot",
        "far-foot", "near-pedal", "far-pedal"
      ].map(id => [id, document.getElementById(id)]));

      let playing = false;
      let manuallyChosen = false;
      let elapsed = 0;
      let previous = null;
      let frame = null;

      function drawLeg(side, angle) {
        const near = side === "near";
        const x = 490 + 31 * Math.cos(angle);
        const y = 426 + 31 * Math.sin(angle);
        const hipX = near ? 514 : 466;
        const hipY = near ? 316 : 307;
        const kneeX = (hipX + x) / 2 + (near ? 23 : -19);
        const kneeY = (hipY + y) / 2 - 1;
        const path = `M${hipX} ${hipY}L${kneeX} ${kneeY}L${x} ${y}`;
        parts[`${side}-leg`].setAttribute("d", path);
        if (near) parts["near-leg-highlight"].setAttribute("d", path);
        parts[`${side}-foot`].setAttribute("d", `M${x - 8} ${y}h26`);
        parts[`${side}-pedal`].setAttribute("d", `M${x - 12} ${y + 8}h29`);
      }

      function render() {
        const angle = elapsed / 1600 * Math.PI * 2;
        drawLeg("near", angle);
        drawLeg("far", angle + Math.PI);
      }

      function tick(now) {
        if (!playing) return;
        if (previous !== null) elapsed += now - previous;
        previous = now;
        render();
        frame = requestAnimationFrame(tick);
      }

      function setPlaying(value) {
        playing = value;
        document.body.classList.toggle("paused", !playing);
        const text = playing ? "暂停动画" : "播放动画";
        label.textContent = text;
        button.setAttribute("aria-label", text);
        previous = null;
        if (frame !== null) cancelAnimationFrame(frame);
        frame = playing ? requestAnimationFrame(tick) : null;
      }

      function toggle() {
        manuallyChosen = true;
        setPlaying(!playing);
      }

      button.addEventListener("click", toggle);
      document.addEventListener("keydown", event => {
        if (event.code !== "Space" || event.altKey || event.ctrlKey || event.metaKey) return;
        const target = event.target;
        if (target instanceof HTMLElement &&
            (target.isContentEditable || /^(INPUT|TEXTAREA|SELECT)$/.test(target.tagName))) return;
        event.preventDefault();
        if (!event.repeat) toggle();
      });

      preference.addEventListener("change", event => {
        if (!manuallyChosen) setPlaying(!event.matches);
      });

      document.addEventListener("visibilitychange", () => {
        previous = null;
      });

      render();
      setPlaying(!preference.matches);
    })();
  </script>
