  <script>
    (() => {
      const button = document.querySelector(".control");
      const label = button.querySelector(".label");
      const reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)");

      const setPlaying = (playing) => {
        document.body.classList.toggle("is-paused", !playing);
        document.body.classList.toggle("is-playing", playing);
        button.setAttribute("aria-pressed", String(playing));
        button.setAttribute("aria-label", playing ? "暂停动画" : "播放动画");
        label.textContent = playing ? "暂停" : "播放";
      };

      setPlaying(!reducedMotion.matches);

      button.addEventListener("click", () => {
        setPlaying(document.body.classList.contains("is-paused"));
      });

      window.addEventListener("keydown", (event) => {
        const isSpace = event.code === "Space" || event.key === " ";
        const editable = event.target instanceof HTMLElement &&
          (event.target.isContentEditable || /^(INPUT|TEXTAREA|SELECT|BUTTON)$/.test(event.target.tagName));
        if (!isSpace || editable) return;
        event.preventDefault();
        setPlaying(document.body.classList.contains("is-paused"));
      });

      reducedMotion.addEventListener("change", (event) => {
        setPlaying(!event.matches);
      });
    })();
  </script>
