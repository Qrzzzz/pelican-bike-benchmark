  <script>
    (() => {
      const root = document.documentElement;
      const button = document.getElementById('motionToggle');
      const prefersReduced = window.matchMedia('(prefers-reduced-motion: reduce)');
      let playing = !prefersReduced.matches;
      function render() {
        root.classList.toggle('is-paused', !playing);
        button.setAttribute('aria-pressed', String(playing));
        button.setAttribute('aria-label', playing ? '暂停动画' : '播放动画');
        button.lastElementChild.textContent = playing ? '暂停动画' : '播放动画';
      }
      function toggle() { playing = !playing; render(); }
      button.addEventListener('click', toggle);
      document.addEventListener('keydown', event => {
        if (event.code === 'Space' && !event.altKey && !event.ctrlKey && !event.metaKey) {
          event.preventDefault(); toggle();
        }
      });
      prefersReduced.addEventListener('change', event => { if (event.matches) { playing = false; render(); } });
      render();
    })();
  </script>
