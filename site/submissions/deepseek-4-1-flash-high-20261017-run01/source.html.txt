<script>
(function () {
  "use strict";

  var app = document.getElementById("app");
  var toggle = document.getElementById("toggle");
  var toggleLabel = document.getElementById("toggle-label");
  var status = document.getElementById("status");
  var rider = document.getElementById("rider");
  var reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)");

  var playing = false;
  var rafId = null;
  var lastTime = 0;
  var crankAngle = 0.62;

  var BB = { x: 590, y: 565 };
  var CRANK_RADIUS = 44;
  var NEAR_HIP = { x: 574, y: 420 };
  var FAR_HIP = { x: 557, y: 418 };
  var NEAR_LEG = { l1: 98, l2: 104 };
  var FAR_LEG = { l1: 96, l2: 102 };

  var el = {
    nearCrankOutline: document.getElementById("near-crank-outline"),
    nearCrank: document.getElementById("near-crank"),
    farCrankOutline: document.getElementById("far-crank-outline"),
    farCrank: document.getElementById("far-crank"),
    nearThighOutline: document.getElementById("near-thigh-outline"),
    nearThigh: document.getElementById("near-thigh"),
    nearShinOutline: document.getElementById("near-shin-outline"),
    nearShin: document.getElementById("near-shin"),
    farThighOutline: document.getElementById("far-thigh-outline"),
    farThigh: document.getElementById("far-thigh"),
    farShinOutline: document.getElementById("far-shin-outline"),
    farShin: document.getElementById("far-shin"),
    nearPedal: document.getElementById("near-pedal"),
    nearFoot: document.getElementById("near-foot"),
    nearKnee: document.getElementById("near-knee"),
    farPedal: document.getElementById("far-pedal"),
    farFoot: document.getElementById("far-foot"),
    farKnee: document.getElementById("far-knee"),
    chainring: document.getElementById("chainring")
  };

  function polarPoint(angle, sign) {
    return {
      x: BB.x + Math.cos(angle) * CRANK_RADIUS * sign,
      y: BB.y + Math.sin(angle) * CRANK_RADIUS * sign
    };
  }

  function solveKnee(hip, ankle, l1, l2) {
    var dx = ankle.x - hip.x;
    var dy = ankle.y - hip.y;
    var rawDistance = Math.hypot(dx, dy) || 0.0001;
    var unitX = dx / rawDistance;
    var unitY = dy / rawDistance;
    var distance = Math.min(l1 + l2 - 0.01, Math.max(Math.abs(l1 - l2) + 0.01, rawDistance));
    var end = {
      x: hip.x + unitX * distance,
      y: hip.y + unitY * distance
    };
    var baseAngle = Math.atan2(unitY, unitX);
    var cosine = (l1 * l1 + distance * distance - l2 * l2) / (2 * l1 * distance);
    var bend = Math.acos(Math.max(-1, Math.min(1, cosine)));
    var kneeAngle = baseAngle - bend;

    return {
      end: end,
      knee: {
        x: hip.x + Math.cos(kneeAngle) * l1,
        y: hip.y + Math.sin(kneeAngle) * l1
      }
    };
  }

  function movePath(node, from, to) {
    node.setAttribute("d", "M " + from.x.toFixed(2) + " " + from.y.toFixed(2) + " L " + to.x.toFixed(2) + " " + to.y.toFixed(2));
  }

  function updateLeg(hip, pedal, lengths, ids) {
    var ankle = { x: pedal.x, y: pedal.y - 7 };
    var solved = solveKnee(hip, ankle, lengths.l1, lengths.l2);
    var lowerEnd = solved.end;
    var knee = solved.knee;

    movePath(ids.thighOutline, hip, knee);
    movePath(ids.thigh, hip, knee);
    movePath(ids.shinOutline, knee, lowerEnd);
    movePath(ids.shin, knee, lowerEnd);
    ids.knee.setAttribute("cx", knee.x.toFixed(2));
    ids.knee.setAttribute("cy", knee.y.toFixed(2));
  }

  function updateCrank(pedal, outline, arm) {
    movePath(outline, BB, pedal);
    movePath(arm, BB, pedal);
  }

  function updatePedal(pedalNode, footNode, pedal) {
    var transform = "translate(" + pedal.x.toFixed(2) + " " + pedal.y.toFixed(2) + ")";
    pedalNode.setAttribute("transform", transform);
    footNode.setAttribute("transform", transform);
  }

  function renderRig(angle) {
    var nearPedal = polarPoint(angle, 1);
    var farPedal = polarPoint(angle, -1);

    updateCrank(nearPedal, el.nearCrankOutline, el.nearCrank);
    updateCrank(farPedal, el.farCrankOutline, el.farCrank);

    updatePedal(el.nearPedal, el.nearFoot, nearPedal);
    updatePedal(el.farPedal, el.farFoot, farPedal);

    updateLeg(NEAR_HIP, nearPedal, NEAR_LEG, {
      thighOutline: el.nearThighOutline,
      thigh: el.nearThigh,
      shinOutline: el.nearShinOutline,
      shin: el.nearShin,
      knee: el.nearKnee
    });

    updateLeg(FAR_HIP, farPedal, FAR_LEG, {
      thighOutline: el.farThighOutline,
      thigh: el.farThigh,
      shinOutline: el.farShinOutline,
      shin: el.farShin,
      knee: el.farKnee
    });

    var ringDegrees = angle * 180 / Math.PI;
    el.chainring.setAttribute("transform", "rotate(" + ringDegrees.toFixed(2) + " 590 565)");
  }

  function step(now) {
    var delta = Math.min(0.05, Math.max(0, (now - lastTime) / 1000));
    lastTime = now;
    crankAngle = (crankAngle + delta * 1.52) % (Math.PI * 2);
    renderRig(crankAngle);
    rafId = window.requestAnimationFrame(step);
  }

  function setPlaying(next, announce) {
    playing = next;
    app.classList.toggle("is-paused", !playing);
    toggleLabel.textContent = playing ? "Pause" : "Play";
    toggle.setAttribute("aria-label", (playing ? "Pause" : "Play") + " animation (Space)");
    toggle.setAttribute("title", (playing ? "Pause" : "Play") + " animation (Space)");

    if (playing) {
      if (rafId === null) {
        lastTime = window.performance.now();
        rafId = window.requestAnimationFrame(step);
      }
    } else if (rafId !== null) {
      window.cancelAnimationFrame(rafId);
      rafId = null;
    }

    if (announce) {
      status.textContent = playing ? "Animation playing" : "Animation paused";
    }
  }

  function layoutRider() {
    var width = window.innerWidth;
    var height = window.innerHeight;

    if (!width || !height) {
      return;
    }

    var sliceScale = Math.max(width / 1200, height / 800);
    var visibleWidth = width / sliceScale;
    var visibleHeight = height / sliceScale;
    var visibleTop = 800 - visibleHeight;
    var riderLeft = 355;
    var riderRight = 925;
    var riderTop = 185;
    var riderGround = 665;
    var riderWidth = riderRight - riderLeft;
    var riderHeight = riderGround - riderTop;

    var scale = Math.min(
      1,
      (visibleWidth * 0.88) / riderWidth,
      (riderGround - visibleTop - 14) / riderHeight
    );

    if (scale > 0.999) {
      rider.removeAttribute("transform");
      return;
    }

    var riderCenter = (riderLeft + riderRight) / 2;
    var translateX = 600 - riderCenter * scale;
    var translateY = riderGround - riderGround * scale;

    rider.setAttribute(
      "transform",
      "translate(" + translateX.toFixed(2) + " " + translateY.toFixed(2) + ") scale(" + scale.toFixed(4) + ")"
    );
  }

  toggle.addEventListener("click", function () {
    setPlaying(!playing, true);
  });

  window.addEventListener("keydown", function (event) {
    var isSpace = event.code === "Space" || event.key === " ";
    if (!isSpace) {
      return;
    }

    var target = event.target;
    if (target && target.closest && target.closest("input, textarea, select, [contenteditable='true']")) {
      return;
    }

    event.preventDefault();
    if (!event.repeat) {
      setPlaying(!playing, true);
    }
  }, { passive: false });

  if (typeof reducedMotion.addEventListener === "function") {
    reducedMotion.addEventListener("change", function (event) {
      if (event.matches) {
        setPlaying(false, true);
      }
    });
  }

  window.addEventListener("resize", layoutRider, { passive: true });
  window.addEventListener("orientationchange", layoutRider, { passive: true });

  renderRig(crankAngle);
  layoutRider();
  setPlaying(!reducedMotion.matches, false);
})();
</script>
