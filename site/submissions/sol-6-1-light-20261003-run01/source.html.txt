<script>
(() => {
  "use strict";
  const get = id => document.getElementById(id);
  const button = get("toggle");
  const preference = window.matchMedia("(prefers-reduced-motion: reduce)");
  const parts = {rider:get("rider"),rear:get("rear-spokes"),front:get("front-spokes"),crank:get("crank"),nearLeg:get("near-leg"),farLeg:get("far-leg"),nearFoot:get("near-foot"),farFoot:get("far-foot"),pedal:get("near-pedal"),clouds:get("clouds"),water:get("water"),road:get("road"),plants:get("plants"),breeze:get("breeze"),shadow:get("shadow")};
  let playing = !preference.matches, manuallyChosen = false, elapsed = 0, previous = null, frame = 0;
  function draw(time) {
    const angle = time * Math.PI * 2 / 1.65, degrees = angle * 180 / Math.PI;
    const x = 559 + Math.cos(angle) * 27, y = 456 + Math.sin(angle) * 27;
    const backX = 559 - Math.cos(angle) * 27, backY = 456 - Math.sin(angle) * 27, bob = Math.sin(angle * 2) * 2;
    parts.rear.setAttribute("transform", `rotate(${degrees} 400 456)`);
    parts.front.setAttribute("transform", `rotate(${degrees} 735 456)`);
    parts.crank.setAttribute("transform", `rotate(${degrees} 559 456)`);
    parts.nearLeg.setAttribute("d", `M563 346 Q587 372 ${594 + (x - 559) * .2} ${396 + (y - 456) * .35} L${x} ${y - 5}`);
    parts.farLeg.setAttribute("d", `M542 354 Q548 378 ${554 + (backX - 559) * .25} ${406 + (backY - 456) * .3} L${backX} ${backY - 5}`);
    parts.nearFoot.setAttribute("d", `M${x - 9} ${y - 5}q13-3 23 5h-23Z`);
    parts.farFoot.setAttribute("d", `M${backX - 9} ${backY - 3}h25`);
    parts.pedal.setAttribute("d", `M${x - 12} ${y + 2}h25`);
    parts.rider.setAttribute("transform", `translate(0 ${bob})`);
    parts.clouds.setAttribute("transform", `translate(${-((time * 8) % 1100)} 0)`);
    parts.road.setAttribute("transform", `translate(${-((time * 105) % 190)} 0)`);
    parts.water.setAttribute("x", -((time * 12) % 220));
    parts.plants.setAttribute("transform", `translate(${-((time * 65) % 1100)} 0)`);
    parts.breeze.setAttribute("transform", `translate(${-Math.sin(angle * .5) * 8} 0)`);
    parts.breeze.setAttribute("opacity", .5 + Math.sin(angle * .5) * .2);
    parts.shadow.setAttribute("rx", 206 - bob * 1.5);
  }
  function tick(timestamp) {
    frame = 0;
    if (!playing || document.hidden) { previous = null; return; }
    if (previous !== null) elapsed += Math.min((timestamp - previous) / 1000, .05);
    previous = timestamp; draw(elapsed); frame = requestAnimationFrame(tick);
  }
  function schedule() { if (playing && !document.hidden && !frame) frame = requestAnimationFrame(tick); }
  function update() {
    button.setAttribute("aria-pressed", String(playing));
    button.setAttribute("aria-label", playing ? "暂停动画" : "播放动画");
    get("control-text").textContent = playing ? "暂停" : "播放";
    get("control-icon").setAttribute("d", playing ? "M4 3h3v10H4zm5 0h3v10H9z" : "M4 2l10 6-10 6z");
    if (frame) cancelAnimationFrame(frame);
    frame = 0; previous = null; schedule();
  }
  function toggle() { manuallyChosen = true; playing = !playing; update(); }
  button.addEventListener("click", toggle);
  document.addEventListener("keydown", event => {
    if (event.code !== "Space" || event.altKey || event.ctrlKey || event.metaKey) return;
    const target = event.target;
    if (target instanceof Element && target.closest("input, textarea, select, [contenteditable='true']")) return;
    event.preventDefault(); if (!event.repeat) toggle();
  });
  document.addEventListener("visibilitychange", () => { if (frame) cancelAnimationFrame(frame); frame = 0; previous = null; schedule(); });
  preference.addEventListener("change", event => { if (!manuallyChosen) { playing = !event.matches; update(); } });
  draw(0); update();
})();
</script>
