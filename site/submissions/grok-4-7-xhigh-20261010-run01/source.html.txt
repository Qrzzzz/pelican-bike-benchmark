<header>
  <h1>鹈鹕骑行</h1>
  <button id="toggle" type="button" aria-pressed="true">暂停</button>

<main>
  <canvas id="stage" role="img" aria-label="一只鹈鹕正沿着海边公路骑自行车前进，车轮和脚踏在转动，海鸥与云朵向后掠过。"></canvas>
  <p id="status" class="visually-hidden" aria-live="polite">动画正在播放</p>
</main>
<script>
(function () {
  "use strict";

  var canvas = document.getElementById("stage");
  var btn = document.getElementById("toggle");
  var status = document.getElementById("status");
  var ctx = canvas.getContext("2d");

  var reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
  var playing = !reduce;
  var raf = 0;
  var last = 0;
  var t = 0;

  var VW = 1200;
  var VH = 800;

  function resize() {
    var dpr = Math.min(window.devicePixelRatio || 1, 2);
    var w = canvas.clientWidth;
    var h = canvas.clientHeight;
    canvas.width = Math.max(1, Math.round(w * dpr));
    canvas.height = Math.max(1, Math.round(h * dpr));
  }

  function syncControl() {
    btn.setAttribute("aria-pressed", playing ? "true" : "false");
    btn.textContent = playing ? "暂停" : "播放";
    status.textContent = playing ? "动画正在播放" : "动画已暂停";
  }

  function toggle() {
    playing = !playing;
    syncControl();
    if (playing) {
      last = 0;
      raf = requestAnimationFrame(frame);
    } else if (raf) {
      cancelAnimationFrame(raf);
      raf = 0;
    }
  }

  btn.addEventListener("click", toggle);
  window.addEventListener("keydown", function (e) {
    if (e.code !== "Space" && e.key !== " ") return;
    var tag = (e.target && e.target.tagName) || "";
    if (tag === "INPUT" || tag === "TEXTAREA") return;
    e.preventDefault();
    toggle();
  });
  window.addEventListener("resize", function () {
    resize();
    draw(t);
  });

  function roundRect(x, y, w, h, r) {
    var rr = Math.min(r, w / 2, h / 2);
    ctx.beginPath();
    ctx.moveTo(x + rr, y);
    ctx.arcTo(x + w, y, x + w, y + h, rr);
    ctx.arcTo(x + w, y + h, x, y + h, rr);
    ctx.arcTo(x, y + h, x, y, rr);
    ctx.arcTo(x, y, x + w, y, rr);
    ctx.closePath();
  }

  function cloud(x, y, s) {
    ctx.save();
    ctx.translate(x, y);
    ctx.scale(s, s);
    ctx.fillStyle = "rgba(255,255,255,0.92)";
    ctx.beginPath();
    ctx.arc(0, 10, 26, 0, Math.PI * 2);
    ctx.arc(32, 4, 34, 0, Math.PI * 2);
    ctx.arc(68, 12, 24, 0, Math.PI * 2);
    ctx.arc(40, 22, 22, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();
  }

  function gull(x, y, s, flap) {
    ctx.save();
    ctx.translate(x, y);
    ctx.scale(s, s);
    ctx.strokeStyle = "#f7fbff";
    ctx.lineWidth = 3;
    ctx.lineCap = "round";
    ctx.beginPath();
    ctx.moveTo(-22, 4);
    ctx.quadraticCurveTo(-10, -16 * flap, 0, 0);
    ctx.quadraticCurveTo(10, -16 * flap, 22, 4);
    ctx.stroke();
    ctx.restore();
  }

  function wheel(x, y, r, spin) {
    ctx.save();
    ctx.translate(x, y);
    ctx.rotate(spin);
    ctx.beginPath();
    ctx.arc(0, 0, r, 0, Math.PI * 2);
    ctx.fillStyle = "#2b2b2b";
    ctx.fill();
    ctx.beginPath();
    ctx.arc(0, 0, r - 7, 0, Math.PI * 2);
    ctx.fillStyle = "#dce7ef";
    ctx.fill();
    ctx.strokeStyle = "#3a4652";
    ctx.lineWidth = 3;
    var i, a;
    for (i = 0; i < 8; i++) {
      a = (i / 8) * Math.PI * 2;
      ctx.beginPath();
      ctx.moveTo(0, 0);
      ctx.lineTo(Math.cos(a) * (r - 8), Math.sin(a) * (r - 8));
      ctx.stroke();
    }
    ctx.beginPath();
    ctx.arc(0, 0, 8, 0, Math.PI * 2);
    ctx.fillStyle = "#f4c95d";
    ctx.fill();
    ctx.restore();
  }

  function leg(hip, foot, L1, L2) {
    var dx = foot.x - hip.x;
    var dy = foot.y - hip.y;
    var dist = Math.hypot(dx, dy);
    var max = L1 + L2 - 0.5;
    var min = Math.abs(L1 - L2) + 0.5;
    dist = Math.max(min, Math.min(max, dist));
    var ang = Math.atan2(dy, dx);
    var cosA = (L1 * L1 + dist * dist - L2 * L2) / (2 * L1 * dist);
    var a = Math.acos(Math.max(-1, Math.min(1, cosA)));
    var a1 = ang + a;
    return {
      x: hip.x + Math.cos(a1) * L1,
      y: hip.y + Math.sin(a1) * L1
    };
  }

  function limb(a, b, c, width, color) {
    ctx.strokeStyle = color;
    ctx.lineWidth = width;
    ctx.lineCap = "round";
    ctx.lineJoin = "round";
    ctx.beginPath();
    ctx.moveTo(a.x, a.y);
    ctx.lineTo(b.x, b.y);
    ctx.lineTo(c.x, c.y);
    ctx.stroke();
  }

  function draw(time) {
    var cw = canvas.width;
    var ch = canvas.height;
    var scale = Math.min(cw / VW, ch / VH);
    var ox = (cw - VW * scale) / 2;
    var oy = (ch - VH * scale) / 2;
    var bob = Math.sin(time * 7.2) * 3.2;
    var spin = time * 9;
    var scroll = time * 220;
    var sky, i, x, y, color, hills, flap, bx, by, R, bb, crank;
    var pedalA, pedalB, footA, footB, px, py, hipA, hipB, kneeA, kneeB, wingLift;

    ctx.setTransform(1, 0, 0, 1, 0, 0);
    ctx.clearRect(0, 0, cw, ch);
    ctx.setTransform(scale, 0, 0, scale, ox, oy);

    sky = ctx.createLinearGradient(0, 0, 0, 470);
    sky.addColorStop(0, "#67b7ea");
    sky.addColorStop(0.55, "#b6e2fb");
    sky.addColorStop(1, "#e7f6ff");
    ctx.fillStyle = sky;
    ctx.fillRect(0, 0, VW, 500);

    ctx.fillStyle = "#f6d98a";
    ctx.beginPath();
    ctx.arc(980, 92, 46, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = "rgba(255,236,170,0.35)";
    ctx.beginPath();
    ctx.arc(980, 92, 70, 0, Math.PI * 2);
    ctx.fill();

    for (i = 0; i < 4; i++) {
      x = ((i * 340 - scroll * 0.25) % (VW + 280)) ;
      if (x < 0) x += VW + 280;
      x -= 140;
      cloud(x, 70 + (i % 2) * 46, 0.85 + (i % 3) * 0.18);
    }

    hills = ["#8fbf6d", "#79ae58", "#6aa34c"];
    for (i = 0; i < hills.length; i++) {
      color = hills[i];
      ctx.fillStyle = color;
      ctx.beginPath();
      ctx.moveTo(0, 500);
      for (x = 0; x <= VW; x += 20) {
        y = 430 + i * 16 + Math.sin((x + scroll * (0.15 + i * 0.08)) / (90 - i * 12)) * (22 - i * 4);
        ctx.lineTo(x, y);
      }
      ctx.lineTo(VW, 500);
      ctx.closePath();
      ctx.fill();
    }

    ctx.fillStyle = "#5f8f42";
    ctx.fillRect(0, 548, VW, 252);
    ctx.fillStyle = "#4d4f55";
    ctx.fillRect(0, 575, VW, 150);

    ctx.strokeStyle = "#f4f1e8";
    ctx.lineWidth = 8;
    ctx.setLineDash([46, 34]);
    ctx.lineDashOffset = -scroll;
    ctx.beginPath();
    ctx.moveTo(0, 648);
    ctx.lineTo(VW, 648);
    ctx.stroke();
    ctx.setLineDash([]);

    ctx.fillStyle = "#6d5438";
    ctx.fillRect(0, 725, VW, 75);
    for (i = 0; i < 8; i++) {
      x = ((i * 180 - scroll * 1.15) % (VW + 80));
      if (x < 0) x += VW + 80;
      x -= 40;
      ctx.fillStyle = i % 2 ? "#d8b48a" : "#c9d7ea";
      ctx.beginPath();
      ctx.ellipse(x, 760, 16, 8, 0, 0, Math.PI * 2);
      ctx.fill();
    }

    flap = 0.55 + Math.sin(time * 8) * 0.45;
    for (i = 0; i < 3; i++) {
      x = ((i * 420 - scroll * 0.55) % (VW + 120));
      if (x < 0) x += VW + 120;
      x -= 60;
      gull(x, 150 + i * 28, 1, flap);
    }

    bx = 430;
    by = 612 + bob;
    R = 78;

    wheel(bx, by, R, spin);
    wheel(bx + 250, by, R, spin);

    ctx.strokeStyle = "#245c86";
    ctx.lineWidth = 9;
    ctx.lineJoin = "round";
    ctx.lineCap = "round";
    ctx.beginPath();
    ctx.moveTo(bx, by);
    ctx.lineTo(bx + 118, by - 92);
    ctx.lineTo(bx + 250, by);
    ctx.moveTo(bx + 118, by - 92);
    ctx.lineTo(bx + 150, by);
    ctx.moveTo(bx + 78, by - 18);
    ctx.lineTo(bx + 168, by - 18);
    ctx.stroke();

    ctx.strokeStyle = "#1b3f5c";
    ctx.lineWidth = 7;
    ctx.beginPath();
    ctx.moveTo(bx + 250, by);
    ctx.lineTo(bx + 286, by - 118);
    ctx.lineTo(bx + 248, by - 132);
    ctx.stroke();
    ctx.fillStyle = "#1b3f5c";
    ctx.beginPath();
    ctx.arc(bx + 248, by - 132, 7, 0, Math.PI * 2);
    ctx.arc(bx + 286, by - 118, 6, 0, Math.PI * 2);
    ctx.fill();

    ctx.fillStyle = "#c4493a";
    roundRect(bx + 92, by - 28, 62, 16, 6);
    ctx.fill();

    bb = { x: bx + 150, y: by };
    crank = 34;
    pedalA = spin;
    pedalB = spin + Math.PI;
    footA = { x: bb.x + Math.cos(pedalA) * crank, y: bb.y + Math.sin(pedalA) * crank };
    footB = { x: bb.x + Math.cos(pedalB) * crank, y: bb.y + Math.sin(pedalB) * crank };

    ctx.strokeStyle = "#d7dee6";
    ctx.lineWidth = 5;
    ctx.beginPath();
    ctx.arc(bb.x, bb.y, 16, 0, Math.PI * 2);
    ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(bb.x, bb.y);
    ctx.lineTo(footA.x, footA.y);
    ctx.moveTo(bb.x, bb.y);
    ctx.lineTo(footB.x, footB.y);
    ctx.stroke();

    px = bx + 92;
    py = by - 168 + bob * 0.35;
    hipA = { x: px + 18, y: py + 118 };
    hipB = { x: px + 2, y: py + 122 };
    kneeA = leg(hipA, footA, 78, 74);
    kneeB = leg(hipB, footB, 78, 74);

    limb(hipB, kneeB, footB, 11, "#f0a05a");
    limb(hipA, kneeA, footA, 12, "#e9893a");

    function pedalPad(p) {
      ctx.save();
      ctx.translate(p.x, p.y);
      ctx.rotate(0.2);
      ctx.fillStyle = "#222";
      roundRect(-16, -5, 32, 10, 3);
      ctx.fill();
      ctx.restore();
    }
    pedalPad(footB);
    pedalPad(footA);

    ctx.fillStyle = "#f4f7fb";
    ctx.beginPath();
    ctx.ellipse(px + 34, py + 78, 58, 46, -0.35, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = "#e7edf3";
    ctx.beginPath();
    ctx.ellipse(px + 18, py + 92, 28, 18, -0.4, 0, Math.PI * 2);
    ctx.fill();

    ctx.fillStyle = "#fff";
    ctx.beginPath();
    ctx.ellipse(px + 8, py + 18, 34, 30, -0.2, 0, Math.PI * 2);
    ctx.fill();

    ctx.fillStyle = "#1d2430";
    ctx.beginPath();
    ctx.arc(px + 18, py + 12, 5.5, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = "#fff";
    ctx.beginPath();
    ctx.arc(px + 20, py + 10, 2, 0, Math.PI * 2);
    ctx.fill();

    ctx.strokeStyle = "#f3b24a";
    ctx.lineWidth = 10;
    ctx.lineCap = "round";
    ctx.beginPath();
    ctx.moveTo(px + 28, py + 20);
    ctx.quadraticCurveTo(px + 120, py + 8, px + 168, py + 34);
    ctx.stroke();
    ctx.fillStyle = "#f0a328";
    ctx.beginPath();
    ctx.moveTo(px + 78, py + 16);
    ctx.quadraticCurveTo(px + 108, py + 48, px + 150, py + 36);
    ctx.quadraticCurveTo(px + 112, py + 28, px + 86, py + 24);
    ctx.fill();

    wingLift = Math.sin(time * 6.5) * 10;
    ctx.fillStyle = "#d5dee8";
    ctx.beginPath();
    ctx.moveTo(px - 8, py + 62);
    ctx.quadraticCurveTo(px - 78, py + 20 + wingLift, px - 108, py + 78 + wingLift);
    ctx.quadraticCurveTo(px - 70, py + 96, px + 6, py + 100);
    ctx.closePath();
    ctx.fill();
    ctx.strokeStyle = "#b7c4d2";
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.moveTo(px - 10, py + 70);
    ctx.quadraticCurveTo(px - 62, py + 48 + wingLift, px - 96, py + 82 + wingLift);
    ctx.stroke();

    ctx.fillStyle = "#f7b24a";
    ctx.beginPath();
    ctx.moveTo(px + 8, py + 150);
    ctx.lineTo(px - 2, py + 168);
    ctx.lineTo(px + 22, py + 164);
    ctx.closePath();
    ctx.fill();
  }

  function frame(now) {
    if (!playing) return;
    if (!last) last = now;
    var dt = Math.min(0.033, (now - last) / 1000);
    last = now;
    t += dt;
    draw(t);
    raf = requestAnimationFrame(frame);
  }

  if (reduce) {
    btn.setAttribute("aria-pressed", "false");
    btn.textContent = "播放";
    status.textContent = "已根据减少动态偏好暂停，可手动播放";
  } else {
    syncControl();
  }
  resize();
  draw(0);
  if (playing) raf = requestAnimationFrame(frame);
})();
</script>
</body>
</html>
