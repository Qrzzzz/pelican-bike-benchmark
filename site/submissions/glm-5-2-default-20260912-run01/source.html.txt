<script>
(function(){
  var scene = document.getElementById('scene');
  var btn   = document.getElementById('toggleBtn');
  var txt   = document.getElementById('btnText');
  var plays = document.querySelectorAll('.i-play');
  var pauses= document.querySelectorAll('.i-pause');

  var reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var playing = !reduced;

  function apply(){
    document.body.classList.toggle('is-paused', !playing);
    btn.setAttribute('aria-pressed', String(playing));
    txt.textContent = playing ? '暂停' : '播放';
    for (var i=0;i<plays.length;i++)  plays[i].style.display  = playing ? 'none' : '';
    for (var j=0;j<pauses.length;j++) pauses[j].style.display = playing ? '' : 'none';
  }

  function toggle(){ playing = !playing; apply(); }

  btn.addEventListener('click', toggle);

  window.addEventListener('keydown', function(e){
    if (e.key === ' ' || e.code === 'Space'){
      if (e.target === btn) return; // let the button handle it natively
      e.preventDefault();
      toggle();
    }
  });

  apply();
})();
</script>
