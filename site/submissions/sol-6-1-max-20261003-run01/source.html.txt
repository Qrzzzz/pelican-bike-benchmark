          <button id="play-toggle" class="play-button" type="button" aria-label="播放骑行动画" aria-controls="ride-scene" aria-keyshortcuts="Space" disabled>
            <svg class="button-icon" viewBox="0 0 16 16" aria-hidden="true">
              <path class="play-symbol" d="m5 2 9 6-9 6Z" fill="currentColor"/>
              <g class="pause-symbol" fill="currentColor"><rect x="3" y="2" width="3" height="12" rx="1"/><rect x="10" y="2" width="3" height="12" rx="1"/></g>

            <span id="button-label">播放骑行</span>

  <script>
    (() => {
      'use strict';

      const byId = id => document.getElementById(id);
      const button = byId('play-toggle');
      const label = byId('button-label');
      const status = byId('motion-status');
      const scene = byId('ride-scene');
      const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
      const narrowScreen = window.matchMedia('(max-width: 700px)');
      const parts = {
        rear: byId('rear-spokes'), front: byId('front-spokes'),
        crank: byId('crank'), bird: byId('pelican'),
        clouds: byId('clouds'), sea: byId('sea-lines'),
        road: byId('road-dashes'), verge: byId('verge'),
        scarf: byId('scarf-tail'), breeze: byId('breeze'),
        nearLeg: byId('near-leg'), nearOutline: byId('near-leg-outline'),
        farLeg: byId('far-leg'), farOutline: byId('far-leg-outline'),
        nearPedal: byId('near-pedal'), farPedal: byId('far-pedal')
      };

      const TAU = Math.PI * 2;
      let playing = false;
      let elapsed = 0;
      let frameId = null;
      let previousTime = null;

      function fitScene() {
        scene.setAttribute('viewBox', narrowScreen.matches ? '285 130 650 530' : '0 0 1200 660');
      }

      // 两段腿长固定，以脚踏的位置求膝关节，避免腿部伸缩或脚踏脱离。
      function legPath(hipX, hipY, footX, footY, upper, lower) {
        const dx = footX - hipX;
        const dy = footY - hipY;
        const distance = Math.hypot(dx, dy);
        const along = (upper * upper - lower * lower + distance * distance) / (2 * distance);
        const across = Math.sqrt(Math.max(0, upper * upper - along * along));
        const kneeX = hipX + (along * dx + across * dy) / distance;
        const kneeY = hipY + (along * dy - across * dx) / distance;
        return `M${hipX.toFixed(2)} ${hipY.toFixed(2)}L${kneeX.toFixed(2)} ${kneeY.toFixed(2)}L${footX.toFixed(2)} ${footY.toFixed(2)}`;
      }

      function render(time) {
        const phase = time * TAU / 2.8;
        const wheelDegrees = (time * 360 / 2.2) % 360;
        const crankDegrees = (phase * 180 / Math.PI) % 360;
        const bob = Math.sin(phase * 2) * 2;
        const pedalX = 38 * Math.cos(phase);
        const pedalY = 38 * Math.sin(phase);
        const nearX = 572 + pedalX;
        const nearY = 524 + pedalY;
        const farX = 572 - pedalX;
        const farY = 524 - pedalY;
        const nearPath = legPath(551, 420 + bob, nearX - 4, nearY - 10, 73, 78);
        const farPath = legPath(559, 420 + bob, farX - 4, farY - 10, 70, 77);

        parts.rear.setAttribute('transform', `rotate(${wheelDegrees.toFixed(2)})`);
        parts.front.setAttribute('transform', `rotate(${wheelDegrees.toFixed(2)})`);
        parts.crank.setAttribute('transform', `rotate(${crankDegrees.toFixed(2)} 572 524)`);
        parts.bird.setAttribute('transform', `translate(0 ${bob.toFixed(2)})`);
        parts.nearLeg.setAttribute('d', nearPath);
        parts.nearOutline.setAttribute('d', nearPath);
        parts.farLeg.setAttribute('d', farPath);
        parts.farOutline.setAttribute('d', farPath);
        parts.nearPedal.setAttribute('transform', `translate(${nearX.toFixed(2)} ${nearY.toFixed(2)})`);
        parts.farPedal.setAttribute('transform', `translate(${farX.toFixed(2)} ${farY.toFixed(2)})`);

        const flutter = Math.sin(time * 4.2) * 4;
        parts.scarf.setAttribute('d', `M628 318C608 ${309 + flutter} 598 ${321 - flutter} 577 ${310 + flutter}L585 ${326 + flutter}C603 ${327 - flutter} 613 322 631 328Z`);
        parts.clouds.setAttribute('transform', `translate(${-(time * 9 % 1600)} 0)`);
        parts.sea.setAttribute('patternTransform', `translate(${-(time * 26 % 240)} 0)`);
        parts.road.setAttribute('patternTransform', `translate(${-(time * (84 * TAU / 2.2) % 200)} 0)`);
        parts.verge.setAttribute('transform', `translate(${-(time * 110 % 1440)} 0)`);
        parts.breeze.setAttribute('transform', `translate(${Math.sin(time * 2) * 5} 0)`);
        parts.breeze.setAttribute('opacity', String(.55 + .25 * Math.sin(time * 2)));
      }

      // 只在可见且播放时请求下一帧；暂停后保留所有部件的当前位置。
      function tick(timestamp) {
        if (!playing || document.hidden) {
          frameId = null;
          previousTime = null;
          return;
        }
        if (previousTime !== null) elapsed += Math.min(timestamp - previousTime, 64) / 1000;
        previousTime = timestamp;
        render(elapsed);
        frameId = requestAnimationFrame(tick);
      }

      function syncFrames() {
        if (frameId !== null) cancelAnimationFrame(frameId);
        frameId = null;
        previousTime = null;
        if (playing && !document.hidden) frameId = requestAnimationFrame(tick);
      }

      function setPlaying(next, preferenceNotice = false) {
        playing = next;
        document.documentElement.dataset.playing = String(playing);
        label.textContent = playing ? '暂停骑行' : '播放骑行';
        button.setAttribute('aria-label', playing ? '暂停骑行动画' : '播放骑行动画');
        status.textContent = playing ? '海风正好，向前中' : preferenceNotice ? '已减少动态 · 可手动播放' : '歇一会，也很好';
        syncFrames();
      }

      button.addEventListener('click', () => setPlaying(!playing));
      document.addEventListener('keydown', event => {
        if (event.code !== 'Space' && event.key !== ' ') return;
        if (event.defaultPrevented || event.isComposing || event.altKey || event.ctrlKey || event.metaKey) return;
        const target = event.target;
        if (target instanceof Element && target.closest('input, textarea, select, [contenteditable], a, summary')) return;
        event.preventDefault();
        if (!event.repeat) setPlaying(!playing);
      });

      document.addEventListener('visibilitychange', syncFrames);
      narrowScreen.addEventListener('change', fitScene);
      reducedMotion.addEventListener('change', event => {
        if (event.matches) setPlaying(false, true);
      });

      fitScene();
      render(0);
      button.disabled = false;
      setPlaying(!reducedMotion.matches, reducedMotion.matches);
    })();
  </script>
